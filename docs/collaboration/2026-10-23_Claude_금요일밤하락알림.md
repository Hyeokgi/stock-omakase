### GAS 코드 — `매크로.gs` 맨 끝에 붙이는 추가 블록

`dispatchWithInputs_`는 [재료 수집 GAS 문서](../운영_2026-10-10_재료수집_GAS아침발사.md)에서 이미 붙인 함수를 그대로 쓴다.

```javascript
// -------------------------------------------------------------
// 🌙 [2026-10-23 신규] 금요일 밤 하락 알림 발사 (금 19:00)
//    깃허브 예약(18:00·19:15)은 백업. 19:40 에 가격을 보고 −3% 이하 종목이 있을 때만 채널로 보낸다.
// -------------------------------------------------------------
function triggerFridayDipAlert(e) {
  dispatchWithInputs_(e, "friday_dip_alert.yml", {});
}

// ⏰ 금요일 19시 트리거만 설치한다. 이 함수의 기존 트리거만 지우고 다시 만든다.
function setupFridayDipAlarm() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === "triggerFridayDipAlert") ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger("triggerFridayDipAlert").timeBased()
    .onWeekDay(ScriptApp.WeekDay.FRIDAY).atHour(19).nearMinute(0).create();
}
```
